\section{Como ejecutar las pruebas}

\subsection{Requisitos}

\begin{itemize}[leftmargin=*,nosep]
  \item Python 3.10 o superior.
  \item Modulo \texttt{PLY} 3.11 o superior: \texttt{pip install ply}.
  \item Para la UI web: PHP y un navegador.
\end{itemize}

\subsection{Modos de ejecucion}

\begin{caja}{Tres formas de probar el compilador}
\begin{enumerate}[leftmargin=*,nosep]
  \item \textbf{UI escritorio:} \texttt{python main.py}
        Editor + paneles de tokens, AST, semantico, TAC, errores.
        Atajo \texttt{F5} ejecuta el analisis; boton "REPORTES HTML"
        genera los archivos en una carpeta a eleccion.
  \item \textbf{UI web:} servir \texttt{index.php} con PHP. Ejecuta
        el codigo en el navegador y llama a \texttt{bridge.py}.
  \item \textbf{CLI / JSON:} \texttt{python bridge.py archivo.programa}
        Imprime el resultado completo en JSON. Con la opcion
        \texttt{--reportes carpeta/} tambien genera los HTMLs.
\end{enumerate}
\end{caja}

\section{Sintaxis del lenguaje}

\begin{lstlisting}[style=tac]
Estructura:       programa { ... }
Tipos:            entero  decimal  cadena  booleano
Control:          si  sino  mientras  hacer_mientras  para
I/O:              imprimir  leer
Logicos:          verdadero  falso  &&  ||  !
\end{lstlisting}

\begin{lstlisting}[style=ml]
programa {
    entero edad = 25;
    decimal salario = 15750.50;
    cadena nombre = "Ana Garcia";
    booleano activo = verdadero;

    si (edad >= 18 && activo) {
        imprimir(nombre);
    } sino {
        imprimir("menor");
    }

    mientras (edad > 0) { edad = edad - 1; }
    para (entero i = 0; i < 5; i = i + 1) {
        imprimir(i);
    }
}
\end{lstlisting}

\section{Pruebas por criterio de la rubrica}

\subsection{Criterio 1: Deteccion de errores semanticos}

\subsubsection{P1.1 --- Variable no declarada}
\begin{lstlisting}[style=ml]
programa { imprimir(noexiste); }
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 1, Col 20: Variable 'noexiste' no declarada
\end{lstlisting}

\subsubsection{P1.2 --- Variable duplicada}
\begin{lstlisting}[style=ml]
programa {
    entero x = 1;
    entero x = 2;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: Variable 'x' ya fue declarada en este ambito (linea 2)
\end{lstlisting}

\subsubsection{P1.3 --- Tipo incompatible en declaracion}
\begin{lstlisting}[style=ml]
programa {
    entero x = "hola";
    booleano b = 5;
    entero y = 3.14;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 2: Asignacion incompatible: variable 'x' de tipo
    'entero' no puede recibir un valor 'cadena'
[Semantico] Linea 3: Asignacion incompatible: variable 'b' de tipo
    'booleano' no puede recibir un valor 'entero'
[Semantico] Linea 4: Asignacion incompatible: variable 'y' de tipo
    'entero' no puede recibir un valor 'decimal'
\end{lstlisting}

\subsubsection{P1.4 --- Widening permitido (no es error)}
\begin{lstlisting}[style=ml]
programa {
    decimal x = 5;
    decimal y = 2 + 3;
}
\end{lstlisting}
\begin{prueba}{Resultado esperado}
Sin errores. La promocion \texttt{entero} $\rightarrow$ \texttt{decimal}
es semanticamente segura y se acepta.
\end{prueba}

\subsubsection{P1.5 --- Condicion no booleana}
\begin{lstlisting}[style=ml]
programa {
    entero x = 5;
    si (x) { imprimir(x); }
    mientras (x + 1) { x = x - 1; }
    para (entero i = 0; i; i = i + 1) { imprimir(i); }
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: La condicion del 'si' debe ser booleana, no 'entero'
[Semantico] Linea 4: La condicion del 'mientras' debe ser booleana, no 'entero'
[Semantico] Linea 5: La condicion del 'para' debe ser booleana, no 'entero'
\end{lstlisting}

\subsubsection{P1.6 --- Division por cero}
\begin{lstlisting}[style=ml]
programa {
    entero a = 10 / 0;
    entero b = 10 % 0;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 2: Division por cero (operador '/')
[Semantico] Linea 3: Division por cero (operador '%')
\end{lstlisting}

\subsubsection{P1.7 --- Operacion aritmetica incompatible}
\begin{lstlisting}[style=ml]
programa {
    booleano b = verdadero;
    entero x = b + 1;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: Operacion aritmetica '+' incompatible entre
    'booleano' y 'entero'
\end{lstlisting}

\subsubsection{P1.8 --- Comparacion incompatible}
\begin{lstlisting}[style=ml]
programa {
    entero x = 5;
    booleano b = x == "hola";
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: Comparacion '==' incompatible entre 'entero' y 'cadena'
\end{lstlisting}

\subsubsection{P1.9 --- Operacion logica entre no booleanos}
\begin{lstlisting}[style=ml]
programa {
    entero a = 1; entero b = 2;
    booleano c = a && b;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: Operacion logica '&&' incompatible entre 'entero' y 'entero'
\end{lstlisting}

\subsubsection{P1.10 --- Operador unario invalido}
\begin{lstlisting}[style=ml]
programa {
    booleano b = verdadero;
    entero x = -b;
    entero y = 5;
    booleano c = !y;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 3: Operador unario '-' incompatible con 'booleano'
[Semantico] Linea 5: Operador '!' incompatible con 'entero'
\end{lstlisting}

\subsubsection{P1.11 --- Lectura de variable no declarada}
\begin{lstlisting}[style=ml]
programa { leer(variable_inexistente); }
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 1, Col 17: Variable 'variable_inexistente' no declarada
\end{lstlisting}

\subsubsection{P1.12 --- Asignacion a variable no declarada}
\begin{lstlisting}[style=ml]
programa { y = 10; }
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 1, Col 12: Variable 'y' no declarada
\end{lstlisting}

\subsubsection{P1.13 --- Uso antes de declaracion}
\begin{lstlisting}[style=ml]
programa {
    imprimir(x);
    entero x = 5;
}
\end{lstlisting}
\begin{lstlisting}[style=err]
[Semantico] Linea 2: Variable 'x' usada antes de ser declarada
    (declarada en linea 3)
\end{lstlisting}

\subsection{Criterio 3: Codigo de Tres Direcciones}

\subsubsection{P3.1 --- Expresion aritmetica}
\begin{lstlisting}[style=ml]
programa {
    entero a = 3;
    entero b = 4;
    entero c = (a + b) * 2;
}
\end{lstlisting}
\begin{lstlisting}[style=tac]
1: a = 3
2: b = 4
3: $t1 = a + b
4: $t2 = $t1 * 2
5: c = $t2
\end{lstlisting}

\subsubsection{P3.2 --- Si/sino}
\begin{lstlisting}[style=ml]
programa {
    entero x = 5;
    si (x > 0) { imprimir(x); } sino { imprimir(0); }
}
\end{lstlisting}
\begin{lstlisting}[style=tac]
1: x = 5
2: $t1 = x > 0
3: ifFalse $t1 goto $L1
4: imprimir x
5: goto $L2
6: $L1:
7: imprimir 0
8: $L2:
\end{lstlisting}

\subsubsection{P3.3 --- Mientras}
\begin{lstlisting}[style=ml]
programa {
    entero i = 0;
    mientras (i < 10) { i = i + 1; }
}
\end{lstlisting}
\begin{lstlisting}[style=tac]
1: i = 0
2: $L1:
3: $t1 = i < 10
4: ifFalse $t1 goto $L2
5: $t2 = i + 1
6: i = $t2
7: goto $L1
8: $L2:
\end{lstlisting}

\subsubsection{P3.4 --- Para y hacer-mientras}
\textbf{Para} genera: init, label, comparacion, ifFalse, cuerpo, update,
goto, label final. \\
\textbf{Hacer-mientras} pone la condicion al final, asi el cuerpo
siempre se ejecuta al menos una vez.

\subsection{Criterio 4: Optimizacion}

\subsubsection{P4.1 --- Constant Folding}
\begin{lstlisting}[style=ml]
programa { entero c = (3 + 4) * 2 - 1; imprimir(c); }
\end{lstlisting}

\textbf{Antes:}
\begin{lstlisting}[style=tac]
$t1 = 3 + 4
$t2 = $t1 * 2
$t3 = $t2 - 1
c = $t3
imprimir c
\end{lstlisting}

\textbf{Despues (reduccion 60\%):}
\begin{lstlisting}[style=tac]
c = 13
imprimir 13
\end{lstlisting}

\subsubsection{P4.2 --- Identidades algebraicas}
\texttt{a + 0} $\rightarrow$ \texttt{a}; \texttt{b * 1} $\rightarrow$
\texttt{b}; \texttt{c * 0} $\rightarrow$ \texttt{0}.

\subsubsection{P4.3 --- Branch Pruning}
\begin{lstlisting}[style=ml]
programa { si (verdadero) { imprimir(1); } sino { imprimir(2); } }
\end{lstlisting}
La condicion siempre verdadera elimina el \texttt{ifFalse}. La rama
\texttt{sino} queda inalcanzable y la pasada de \emph{Unreachable Code
Elimination} la elimina.

\subsubsection{P4.4 --- Punto fijo}

El optimizador aplica las 6 pasadas en orden y repite el ciclo hasta
que ninguna haga cambios:
\begin{enumerate}[leftmargin=*,nosep]
  \item Constant Folding \& Algebraic
  \item Constant / Copy Propagation
  \item Branch Pruning
  \item Unreachable Code Elimination
  \item Dead-Code Elimination
  \item Jump Threading
\end{enumerate}
